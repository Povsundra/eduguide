basic_information:
  curriculum_year: "Academic Year 2022–2023"
  program: Bachelor of Engineering in Architectural Engineering
  university: Institute of Technology of Cambodia (ITC)
  faculty: Faculty of Civil Engineering
  department: Department of Architectural Engineering (GAR/DAE)
  duration: 5 years
  foundation_years: 2
  specialization_years: 3
  total_credits: 165
  specialization_credits: 100

  curriculum_overview:

  educational_structure:
    - Common Engineering Foundation Program (Years 1–2)
    - Third Year (GAR/DAE Core Curriculum)
    - Fourth Year (Advanced Architectural Engineering)
    - Fifth Year (Professional Practice & Graduation)
    - Architectural Design Studios
    - Laboratory-based Learning
    - Industrial Internships
    - Graduation Thesis

  learning_progression:

    foundation: >
      Students complete the Common Engineering Foundation Program (Tronc Commun)
      during the first two years before entering the Department of Architectural
      Engineering (GAR/DAE).

    specialization: >
      Beginning in Year 3, students study architectural design, Khmer and world
      architecture, construction materials, structural engineering, surveying,
      computer-aided architectural design, and design studios.

    concentration: >
      During Years 4 and 5 students focus on advanced architectural design,
      urban planning, landscape architecture, reinforced concrete, steel
      structures, professional practice, project management, and graduation
      thesis.

    experiential_learning: >
      The curriculum combines lectures, architectural design workshops,
      computer laboratories, field practice, two compulsory internships,
      and a graduation thesis.

year_structure:

  year_1:
    learning_objectives:
      - Build strong foundations in mathematics, science, engineering, and communication.
      - Prepare students for engineering specialization through the Common Engineering Foundation Program.

    core_topics:
      - Mathematics
      - Physics
      - Chemistry
      - Fundamentals of Information Technology
      - Technical Drawing
      - Environment
      - Marketing
      - Management and Accounting
      - Philosophy
      - History
      - French
      - English

  year_2:
    learning_objectives:
      - Continue the Common Engineering Foundation Program.
      - Strengthen analytical thinking and engineering fundamentals before entering Architectural Engineering.

    core_topics:
      - Mathematics
      - Physics
      - Chemistry
      - Fundamentals of Information Technology
      - Technical Drawing
      - Environment
      - Marketing
      - Management and Accounting
      - Philosophy
      - History
      - French
      - English

  year_3:
    learning_objectives:
      - Develop architectural design thinking and engineering fundamentals.
      - Build technical drawing and building design skills.

    core_topics:
      - Architectural Design Studio
      - Architectural Graphics
      - History of Architecture
      - Building Construction
      - Building Materials
      - Structural Fundamentals
      - Computer-Aided Design (CAD)
      - Architectural Model Making

  year_4:
    learning_objectives:
      - Apply architectural engineering principles to complex building projects.
      - Integrate environmental, structural, and urban design concepts.

    core_topics:
      - Advanced Architectural Design Studio
      - Urban Planning
      - Environmental Design
      - Sustainable Architecture
      - Building Services
      - Structural Design
      - Construction Management
      - Building Information Modeling (BIM)

  year_5:
    learning_objectives:
      - Apply professional architectural engineering practice.
      - Complete internship and graduation design project.

    core_topics:
      - Professional Practice
      - Project Management
      - Architectural Project Development
      - Construction Administration
      - Industrial Internship
      - Graduation Project
